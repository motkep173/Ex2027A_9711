\section*{רמזים}
\addcontentsline{toc}{section}{רמזים}
כאן מופיעים רק הרמזים, בלי הפתרונות. נסו קודם לפתור בעזרת הרמז, ורק אחר כך עברו לפתרון.


\subsection*{מספרים ממשיים}
\noindent \bknum{1}\ (א)\enskip כתבו כל מספר רציונלי כמנה של שני שלמים, ובדקו מה מתקבל בכל פעולה.\par\smallskip
\noindent \bknum{1}\ (ב)\enskip הניחו בשלילה שהתוצאה רציונלית, והשתמשו בסעיף הקודם.\par\smallskip
\noindent \bknum{1}\ (ג)\enskip חפשו שני מספרים אי-רציונליים שהחלק האי-רציונלי שלהם מתבטל בסכום.\par\smallskip
\noindent \bknum{2}\ (א)\enskip חקו את ההוכחה ש-$\sqrt2\notin\mathbb{Q}$.\par\smallskip
\noindent \bknum{2}\ (ב)\enskip עקבו אחרי ההדרכה. כדי להראות ש-$\sqrt2+\sqrt3$ ו-$\sqrt2-\sqrt3$ אינם שניהם רציונליים, הסתכלו על הסכום שלהם.\par\smallskip
\noindent \bknum{3}\ (ג)\enskip בחרו מספר עשרוני סופי קטן מ-$1.415$, והראו שהוא גדול מ-$\sqrt2$ בעזרת העלאה בריבוע.\par\smallskip
\noindent \bknum{5}\ (ב)\enskip כתבו את אי-השוויון כאי-שוויון כפול עבור $\frac{1}{x}$, ורק אחר כך עברו ל-$x$ (שימו לב לסימן לפני שהופכים).\par\smallskip
\noindent \bknum{5}\ (ג)\enskip חלקו את הישר לתחומים לפי הנקודות שבהן הביטויים שבתוך הערכים המוחלטים מחליפים סימן, ופתרו בכל תחום בנפרד.\par\smallskip
\noindent \bknum{5}\ (ד)\enskip מתי מתקיים $|a|>a$? השתמשו בתשובה עם $a=x^2-x-6$.\par\smallskip
\noindent \bknum{6}\ (ב)\enskip היעזרו בסעיף א׳.\par\smallskip
\noindent \bknum{7}\ (ב)\enskip בשלב המעבר השוו את $\frac{2^{n+1}}{(n+1)^3}$ ל-$\frac{2^n}{n^3}$, והשתמשו ב-$n\geq10$ כדי לחסום את היחס $\frac{n^3}{(n+1)^3}$ מלמטה.\par\smallskip
\noindent \bknum{7}\ (ג)\enskip בשלב המעבר כתבו $(n+1)!=n!\cdot(n+1)$ והשתמשו בהנחת האינדוקציה.\par\smallskip
\noindent \bknum{7}\ (ה)\enskip נסחו את הטענה כהצגה $n=3a+5b$ עם $a,b$ שלמים אי-שליליים. בשלב המעבר החליפו כמה מטבעות בהצגה של $n$ כך שהסכום יגדל באחד, ופצלו למקרים לפי סוגי המטבעות שיש בה.\par\smallskip

\subsection*{סדרות}
\noindent \bknum{9}\ (א)\enskip התחילו מהסדרה $(-1)^n$ והפעילו עליה פעולות חשבון פשוטות.\par\smallskip
\noindent \bknum{9}\ (ב)\enskip שימו לב לסימן המתחלף, ובדקו פי כמה גדלים המונה והמכנה מאיבר לאיבר.\par\smallskip
\noindent \bknum{10}\enskip לכל מספר פתרו את המשוואה שבה $a_n$ שווה לו, ובדקו אילו מהפתרונות הם מספרים טבעיים.\par\smallskip
\noindent \bknum{11}\ (א)\enskip פשטו את המרחק בין איבר הסדרה לגבול לשבר אחד, ובדקו עבור אילו $n$ הוא קטן מ-$\varepsilon$.\par\smallskip
\noindent \bknum{11}\ (ג)\enskip חסמו את הביטוי מלמעלה בביטוי פשוט יותר: השתמשו ב-$|\sin n|\le1$ והקטינו את המכנה.\par\smallskip
\noindent \bknum{11}\ (ד)\enskip פשטו את המרחק בין איבר הסדרה ל-$\frac52$ לשבר אחד. כדי לקבל בחירה פשוטה של $N$ אפשר לחסום אותו מלמעלה בביטוי מהצורה $\frac{c}{n}$.\par\smallskip
\noindent \bknum{11}\ (ה)\enskip נסחו את שלילת הגדרת הגבול, ופשטו את המרחק בין איבר הסדרה ל-$\frac32$: האם הוא נשאר רחוק מאפס עבור $n$ גדולים?\par\smallskip
\noindent \bknum{12}\ (א)\enskip בצעד האינדוקציה כתבו $2^{n+1}=2\cdot2^n$ והשתמשו בהנחת האינדוקציה.\par\smallskip
\noindent \bknum{12}\ (ב)\enskip השתמשו בסעיף הקודם כדי לחסום את $\frac{1}{2^n}$ מלמעלה בביטוי פשוט יותר.\par\smallskip
\noindent \bknum{13}\ (א)\enskip השתמשו באי-שוויון המשולש ההפוך $||x|-|y||\le |x-y|$.\par\smallskip
\noindent \bknum{13}\ (ב)\enskip חשבו על סדרה שהערך המוחלט שלה מתכנס אבל הסימן שלה משתנה.\par\smallskip
\noindent \bknum{13}\ (ד)\enskip נסו לחשוב על סדרה שלא מתכנסת, אבל העלאה שלה בריבוע „מסדרת” אותה והופכת אותה למתכנסת.\par\smallskip
\noindent \bknum{16}\ (א)\enskip השתמשו באי-שוויון המשולש.\par\smallskip
\noindent \bknum{16}\ (ב)\enskip היעזרו בכך ש-$|xy|=|x|\cdot|y|$.\par\smallskip
\noindent \bknum{16}\ (ג)\enskip מה קורה למנה כשהמכנה מתקרב לאפס?\par\smallskip
\noindent \bknum{16}\ (ד)\enskip מה ההבדל בין התנאי כאן לתנאי בסעיף הקודם? מה הוא מונע מהמכנה לעשות?\par\smallskip
\noindent \bknum{17}\ (א)\enskip בדקו אם ההתנהגות הלא-מתכנסת של שתי סדרות יכולה להתבטל בסכום. אם נראה לכם שהטענה לא נכונה, חפשו דוגמה נגדית פשוטה.\par\smallskip
\noindent \bknum{17}\ (ב)\enskip חשבו מה הייתם יכולים להסיק על $b_n$ אילו $a_n+b_n$ הייתה מתכנסת, בעזרת אריתמטיקה של גבולות. אם נראה לכם שהטענה לא נכונה, חפשו דוגמה נגדית פשוטה.\par\smallskip
\noindent \bknum{17}\ (ג)\enskip מהי השלילה של ״לפחות אחת מהסדרות מתכנסת״?\par\smallskip
\noindent \bknum{17}\ (ד)\enskip נסחו את הטענה השקולה לה (קונטרה-פוזיטיב) ובדקו אותה בעזרת אריתמטיקה של גבולות.\par\smallskip
\noindent \bknum{18}\ (א)\enskip יש כאן הפרש שורשים מהצורה $\infty-\infty$: הכפילו וחלקו בצמוד.\par\smallskip
\noindent \bknum{18}\ (ב)\enskip כתבו את הסדרה כמכפלה של $\sin n$ בסדרה אחרת. מה ידוע על $\sin n$, ולאן שואפת הסדרה האחרת?\par\smallskip
\noindent \bknum{18}\ (ד)\enskip זהו את המחובר הדומיננטי, חסמו את הביטוי מלמעלה ומלמטה בעזרתו והשתמשו בכלל הסנדוויץ'.\par\smallskip
\noindent \bknum{18}\ (ו)\enskip חסמו את הסכום מלמעלה ומלמטה על ידי החלפת כל המחוברים במחובר הגדול ביותר ובקטן ביותר, והשתמשו בכלל הסנדוויץ'.\par\smallskip
\noindent \bknum{18}\ (ז)\enskip מה הגורם הדומיננטי במונה? ובמכנה?\par\smallskip
\noindent \bknum{19}\enskip חסמו את המכנה מלמטה על ידי החלפת כל מחובר במחובר הקטן ביותר.\par\smallskip
\noindent \bknum{20}\ (א)\enskip זהו מצב $\infty-\infty$ עם הפרש שורשים: הכפילו וחלקו בצמוד.\par\smallskip
\noindent \bknum{20}\ (ב)\enskip שני המחוברים שואפים לאינסוף. כנסו למכנה משותף לפני חישוב הגבול.\par\smallskip
\noindent \bknum{20}\ (ג)\enskip חלקו את המונה והמכנה ב-$n$, ושימו לב שהסדרה $\cos(n)$ חסומה.\par\smallskip
\noindent \bknum{22}\ (ב)\enskip כתבו את האיברים הראשונים, והשוו בין איברים סמוכים לפי הזוגיות של $n$.\par\smallskip
\noindent \bknum{22}\ (ד)\enskip היעזרו ב-$0\le x-\lfloor x\rfloor<1$. למונוטוניות, בדקו את ערכי הסדרה כאשר $n$ הוא ריבוע שלם.\par\smallskip
\noindent \bknum{23}\ (א)\enskip הסתכלו על $a_{n+1}-a_n$.\par\smallskip
\noindent \bknum{23}\ (ב)\enskip השתמשו בזהות $(a-b)^2=a^2-2ab+b^2\ge0$.\par\smallskip
\noindent \bknum{24}\ (א)\enskip הוכיחו באינדוקציה מונוטוניות וחסימות. כדי לנחש חסם, הניחו שהגבול $L$ קיים והציבו אותו בשני אגפי נוסחת הנסיגה.\par\smallskip
\noindent \bknum{24}\ (ב)\enskip לחסימות, נסו להוכיח באינדוקציה חסם כללי $M$ ובדקו איזה תנאי על $M$ נדרש כדי שצעד האינדוקציה יעבוד.\par\smallskip
\noindent \bknum{24}\ (ג)\enskip מצאו את המספר שעבורו $a_{n+1}=a_n$, ופצלו למקרים לפי מיקומו של $a_1$ ביחס אליו. בכל מקרה בדקו מונוטוניות וחסימות.\par\smallskip
\noindent \bknum{25}\ (א)\enskip השתמשו באינדוקציה ובאי-שוויון הממוצעים $\sqrt{xy}\le\frac{x+y}{2}$ לכל $x,y>0$.\par\smallskip
\noindent \bknum{25}\ (ג)\enskip השאיפו את $n$ לאינסוף בשני האגפים של נוסחת הנסיגה של $a_{n+1}$.\par\smallskip
