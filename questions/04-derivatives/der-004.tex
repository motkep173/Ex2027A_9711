\textbf{מצאו את משוואת הישר המשיק לגרף הפונקציה \(f(x)\) בנקודה \(x = x_0\).}
זכרו כי
$f'(x_0)$
שווה לשיפוע של הישר המשיק לגרף הפונקציה של
$f$
בנקודה
$x_0$
\bE\bMs{2}
\item$ x_0 = 3, \quad f(x) = \frac{x}{x+1}$
\item$ x_0 = 1, \quad f(x) = \sqrt{2x^4 - 1}$
\item$ x_0 = \frac{1}{2e}, \quad f(x) = \ln(2x)$
\eMs
\eE

\ifsol
\begin{solbox}
משוואת הישר המשיק לגרף של פונקציה גזירה
$f$
בנקודה
$x_0$
נתונה ע"י
$$.y-f(x_0)=f'(x_0)(x-x_0)$$
\bE
\item

מתקיים
$$.f'(x)=\left(\frac{x}{x+1}\right)'=\frac{1}{(x+1)^2}$$
מכאן
$f'(3)=\frac{1}{16}$,
לכן משוואת הישר המשיק בנקודה
$x_0=3$
נתונה ע"י
$$,y-\frac{3}{4}=\frac{1}{16}(x-3)$$
כלומר,
$$.y=\frac{1}{16}x+\frac{9}{16}$$
\item
בדומה לסעיף הקודם, נגזור
$$,f'(x)=\frac{4x^3}{\sqrt{2x^4-1}}$$
מכאן
$f'(1)=4$,
לכן משוואת הישר המשיק בנקודה
$x_0=1$
נתונה ע"י
$$,y-1=4(x-1)$$
כלומר,
$$.y=4x-3$$
\item
נגזור
$$,f'(x)=\left(\ln(2x)\right)'=\left(\ln2+\ln x\right)'=\frac{1}{x}$$
מכאן
$f'(\tfrac{1}{2e})=2e$,
לכן משוואת הישר המשיק נתונה ע"י
$$,y-(-1)=2e(x-\frac{1}{2e})$$
כלומר,
$$.y=2ex-2$$
\eE
\end{solbox}
\fi
