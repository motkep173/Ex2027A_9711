\textbf{השתמשו בקירוב לינארי לפונקציה המתאימה על מנת לקרב את הביטויים הבאים (בתשובתכם,
אל תסתמכו על חישובים המשתמשים במחשבון):}
\bE\bMs{2}
\item$\ln(1.05)$
\hint{בחרו פונקציה ונקודה קרובה לביטוי שבה קל לחשב את ערך הפונקציה ואת נגזרתה, והשתמשו בקירוב $f(x_0+h)\approx f(x_0)+f'(x_0)h$.}
\begin{solbox}
\unverified
נשתמש בקירוב הלינארי
$f(x_0+h)\approx f(x_0)+f'(x_0)h$
עבור
$f(x)=\ln x$
סביב
$x_0=1$
ועם
$h=0.05$.
מתקיים
$f(1)=0$
ו-
$f'(x)=\frac1x$,
כלומר
$f'(1)=1$.
לכן
$$.\ln(1.05)\approx 0+1\cdot 0.05=0.05$$
\end{solbox}
\item$ \sqrt[3]{1.02}$
\begin{solbox}
\unverified
נשתמש בקירוב הלינארי עבור
$f(x)=\sqrt[3]{x}=x^{\frac13}$
סביב
$x_0=1$
ועם
$h=0.02$.
מתקיים
$f(1)=1$
ו-
$f'(x)=\frac{1}{3}x^{-\frac23}$,
כלומר
$f'(1)=\frac13$.
לכן
$$.\sqrt[3]{1.02}\approx 1+\frac{1}{3}\cdot 0.02=1+\frac{1}{150}\approx 1.0067$$
\end{solbox}
\eMs
\eE
