לכל אחת מהפונקציות
$f_1,f_2,f_3$
הנתונות ע"י
$$,f_1(x)=\case{ccc}{x\sin(\tfrac{1}{x}) & x\neq 0 \\ 0 & x=0}$$
$$,f_2(x)=\case{ccc}{x^2\sin(\tfrac{1}{x}) & x\neq 0 \\ 0 & x=0}$$
$$,f_3(x)=\case{ccc}{x^2\sin(\tfrac{1}{x})-x & x\neq 0 \\ 0 & x=0}$$
קבעו האם היא גזירה ב-
$x=0$,
ואם כן, חשבו את הנגזרת.

\ifsol
\hint{רשמו לכל פונקציה את מנת ההפרשים ב-$0$ לפי ההגדרה, פשטו, ובדקו אם יש לה גבול כאשר $h\to0$.}
\begin{solbox}
מתקיים
\begin{align*}
    f'_1(0)&=\ \lim_{h\to 0}\frac{f_1(0+h)-f_1(0)}{h}\\&=\ \lim_{h\to 0}\frac{h\sin(\tfrac{1}{h})}{h}\\
    &=\ \lim_{h\to 0}\sin(\tfrac{1}{h}),
\end{align*}
בדף הקודם כבר נכחנו לדעת כי
$\displaystyle\lim_{x\to 0}\cos(\tfrac{1}{x})$
לא קיים
(ההוכחה במקרה שלנו דומה),
לכן
$f_1$
לא גזירה ב-
$x=0$.

נמשיך
\begin{align*}
    f'_2(0)&=\ \lim_{h\to 0}\frac{f_2(0+h)-f_1(0)}{h}\\&=\ \lim_{h\to 0}\frac{h^2\sin(\tfrac{1}{h})}{h}\\
    &=\ \lim_{h\to 0}h\sin(\tfrac{1}{h})\\
    &=\ 0,
\end{align*}
כמכפלה של פונקציה אפסה בפונקציה חסומה,
לכן
$f_2$
גזירה ב-
$x=0$
ו-
$f_2'(0)=0$.

ולבסוף
\begin{align*}
    f'_3(0)&=\ \lim_{h\to 0}\frac{f_3(0+h)-f_1(0)}{h}\\&=\ \lim_{h\to 0}\frac{h^2\sin(\tfrac{1}{h})-h}{h}\\
    &=\ \lim_{h\to 0}\left(h\sin(\tfrac{1}{h})-1\right)\\
    &=\ -1,
\end{align*}

לכן
$f_3$
גזירה ב-
$x=0$
ו-
$f'_3(0)=-1$.
\end{solbox}
\fi
