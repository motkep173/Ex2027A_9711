מצאו פונקציה קדומה לפונקציה
$\ds f(x)=\frac{1}{x(x + 1)^2}$
\hint{פרקו לשברים חלקיים, ושימו לב לגורם החוזר במכנה.}
\begin{solbox}
\unverified
נפרק לשברים חלקיים:
$$,\frac{1}{x(x+1)^2}=\frac{A}{x}+\frac{B}{x+1}+\frac{C}{(x+1)^2}$$
כלומר
$1=A(x+1)^2+Bx(x+1)+Cx$.
הצבת
$x=0$
נותנת
$A=1$,
הצבת
$x=-1$
נותנת
$C=-1$,
והשוואת המקדם של
$x^2$
נותנת
$A+B=0$,
כלומר
$B=-1$.
לכן
\begin{align*}
\int\frac{dx}{x(x+1)^2}=\int\lr{\frac1x-\frac{1}{x+1}-\frac{1}{(x+1)^2}}dx=\ln|x|-\ln|x+1|+\frac{1}{x+1}+C,
\end{align*}
כלומר
\begin{align*}
\boxed{F(x)=\ln\abs{\frac{x}{x+1}}+\frac{1}{x+1}+C}
\end{align*}
\end{solbox}
