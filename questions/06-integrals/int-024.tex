מצאו פונקציה קדומה לפונקציה
$\ds f(x)=\frac{1}{x(\ln x)^2}$
\begin{solbox}
יש כאן ביטוי עם
$\ln x$
ונשים לב בנוסף גם כי
$\frac{1}{x}$
נוכח. לכן
נציב $u=\ln x$,
נקבל
$du=\frac{dx}{x}$
ולכן
\begin{align*}
    \int\frac{1}{x(\ln x)^2}&=\ \int\frac{du}{u^2}\\
    &=\ -\frac{1}{u}+C\\
    &=\ -\frac{1}{\ln x}+C.
\end{align*}
\end{solbox}
