חשבו את האינטגרל
$\ds \int x^{2}e^{-2x}\,dx$
\hint{השתמשו באינטגרציה בחלקים פעמיים, ובכל פעם גזרו את החזקה של $x$.}
\begin{solbox}
נשתמש באינטגרציה בחלקים פעמיים.
תחילה נסמן
$$,u=x^2,\ v'=e^{-2x}$$
מכאן
$$,u'=2x,\ v=-\frac{1}{2}e^{-2x}$$
לכן
\begin{align*}
\int x^2 e^{-2x}\,dx
&=-\frac12 x^2 e^{-2x}+\int x e^{-2x}\,dx
\end{align*}

כעת שוב נבצע אינטגרציה בחלקים ב-  $\ds \int x e^{-2x}\,dx$:
$$,u=x,\ v'=e^{-2x}$$
מכאן
$$,u'=1,\ v=-\frac{1}{2}e^{-2x}$$
לכן
\begin{align*}
\int x e^{-2x}\,dx
&=-\frac12 x e^{-2x}-\frac14 e^{-2x}+C.
\end{align*}
בסה"כ קיבלנו
\begin{align*}
\int x^2 e^{-2x}\,dx
&=-\frac12 x^2 e^{-2x}-\frac12 x e^{-2x}-\frac14 e^{-2x}+C\\
&=-\frac{e^{-2x}}{4}\lr{2x^2+2x+1}+C
\end{align*}

\end{solbox}
