חשבו את האינטגרל
$\ds \int \frac{dx}{x^{3}+1}$
\hint{פרקו את המכנה כסכום חזקות שלישיות ופרקו לשברים חלקיים. בחלק עם הגורם הריבועי הפרידו את נגזרת המכנה והשלימו לריבוע.}
\begin{solbox}
\unverified
נפרק את המכנה:
$x^3+1=(x+1)(x^2-x+1)$,
כאשר
$x^2-x+1$
אי-פריק (הדיסקרימיננטה שלילית). נחפש פירוק לשברים חלקיים
$$,\frac{1}{x^3+1}=\frac{A}{x+1}+\frac{Bx+C}{x^2-x+1}$$
כלומר
$1=A(x^2-x+1)+(Bx+C)(x+1)$.
הצבת
$x=-1$
נותנת
$3A=1$,
כלומר
$A=\frac13$.
השוואת המקדם של
$x^2$
נותנת
$A+B=0$,
כלומר
$B=-\frac13$,
והשוואת המקדם החופשי נותנת
$A+C=1$,
כלומר
$C=\frac23$.
לכן
$$.\frac{1}{x^3+1}=\frac{1}{3}\cdot\frac{1}{x+1}-\frac13\cdot\frac{x-2}{x^2-x+1}$$
לאינטגרל השני נכתוב את המונה בעזרת הנגזרת של המכנה
$(2x-1)$
ונשלים לריבוע
$x^2-x+1=\lr{x-\frac12}^2+\frac34$:
\begin{align*}
\int\frac{x-2}{x^2-x+1}\,dx
&=\frac12\int\frac{2x-1}{x^2-x+1}\,dx-\frac32\int\frac{dx}{\lr{x-\frac12}^2+\frac34}\\
&=\frac12\ln\lr{x^2-x+1}-\frac32\cdot\frac{2}{\sqrt3}\arctan\lr{\frac{2x-1}{\sqrt3}}+C_1\\
&=\frac12\ln\lr{x^2-x+1}-\sqrt3\arctan\lr{\frac{2x-1}{\sqrt3}}+C_1.
\end{align*}
לסיכום
\begin{align*}
\boxed{\int\frac{dx}{x^3+1}=\frac13\ln|x+1|-\frac16\ln\lr{x^2-x+1}+\frac{1}{\sqrt3}\arctan\lr{\frac{2x-1}{\sqrt3}}+C}
\end{align*}
\end{solbox}
