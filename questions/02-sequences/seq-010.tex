\textbf{חשבו את גבול הסדרות הבאות ונמקו את צעדיכם.}
\begin{enumerate}\bMs{2}
\item $\ds\lim_{n \to \infty} \left( \sqrt{n^2 - 2n - 1} - \sqrt{n^2 + 2n - 1} \right)$
\hint{זהו מצב $\infty-\infty$ עם הפרש שורשים: הכפילו וחלקו בצמוד.}
\begin{solbox}
    נסמן את הסדרה ב-
$a_n$.
על ידי הכפלה בצמוד נקבל
\begin{align*}
    a_n
    &=\ \left( \sqrt{n^2 - 2n - 1} - \sqrt{n^2 + 2n - 1} \right)\cdot{\color{red}\frac{\sqrt{n^2 - 2n - 1} + \sqrt{n^2 + 2n - 1}}{\sqrt{n^2 - 2n - 1} + \sqrt{n^2 + 2n - 1}}}\\
    &=\ \frac{-4n}{n\sqrt{1 - \frac{2}{n} - \frac{1}{n^2}} + n\sqrt{1 + \frac{2}{n} - \frac{1}{n^2}}}
    =\ \frac{-4}{\sqrt{1 - \frac{2}{n} - \frac{1}{n^2}} + \sqrt{1 + \frac{2}{n} - \frac{1}{n^2}}},
\end{align*}
ע"פ אריתמטיקה של גבולות נסיק כי גבולה של הסדרה הנתונה הוא
$-2$.
\end{solbox}
%%%%%
\item $\ds\lim_{n \to \infty} \left( \frac{n^3+1}{2n^2 - n} - \frac{n^2}{2n + 3} \right)$
\hint{שני המחוברים שואפים לאינסוף. כנסו למכנה משותף לפני חישוב הגבול.}
\begin{solbox}
נשים לב כי זהו מצב של חיסור סדרות ששתיהן שואפות לאינסוף, לכן אריתמטיקה במובן הרחב לא תופסת כאן באופן מיידי.
נציג
\begin{align*}
\frac{n^3+1}{2n^2 - n} - \frac{n^2}{2n + 3}&=\ \frac{(n^3+1)(2n+3)-n^2(2n^2-n)}{(2n^2-n)(2n+3)}\\
&=\ \frac{4n^3+2n+3}{4n^3+4n^2-3n}=\ \frac{4+\frac{2}{n^2}+\frac{3}{n^3}}{4+\frac{4}{n}-\frac{3}{n^2}},
\end{align*}
מאריתמטיקה של גבולות נסיק כי גבולה של הסדרה הנתונה הוא
$1$.
\end{solbox}

%%%%%
\item $\ds\lim_{n \to \infty} \frac{n + \cos(n)}{\sqrt{n^2 + 5n + 1}}$
\hint{חלקו את המונה והמכנה ב-$n$, ושימו לב שהסדרה $\cos(n)$ חסומה.}
\begin{solbox}
נחלק בחזקה הדומיננטית במכנה ונקבל
$$.\frac{n + \cos(n)}{\sqrt{n^2 + 5n + 1}}=\frac{n + \cos(n)}{n\sqrt{1 + \frac{5}{n} + \frac{1}{n^2}}}=\frac{1 + \frac{\cos(n)}{n}}{\sqrt{1 + \frac{5}{n} + \frac{1}{n^2}}}$$
הסדרה $\cos(n)$ חסומה:
$$,-1\le\cos (n)\le 1\qquad\forall n\in\mathbb{N}$$
לכן
$\ds\lim_{n\to\infty}\frac{\cos (n)}{n}=0$
כמכפלה של סדרה חסומה באפסה.
במכנה, $1 + \frac{5}{n} + \frac{1}{n^2}\goes1$, ולפי אריתמטיקה של חזקות גם $\sqrt{1 + \frac{5}{n} + \frac{1}{n^2}}=\lr{1 + \frac{5}{n} + \frac{1}{n^2}}^{1/2}\goes1$.
לכן, מאריתמטיקה של גבולות, נסיק כי
$$.\ds\lim_{n \to \infty} \frac{n + \cos(n)}{\sqrt{n^2 + 5n + 1}}=1$$
\end{solbox}
%%%%
\eMs
\end{enumerate}
