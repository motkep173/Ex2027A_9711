\textbf{רשמו חמישה איברים ראשונים של הסדרה
$(a_n)_{n=1}^{\infty}$
כאשר}
\begin{enumerate}\bMs{2}
\item $\ds .a_n=\frac{n^3-1}{n^2+1}$
\begin{solbox}
הסדרה נתונה על ידי
$a_n=\frac{n^3-1}{n^2+1}$
ולכן
$$.a_1=\frac{1^3-1}{1^2+1}=0\ ,\ a_2=\frac{2^3-1}{2^2+1}=\frac75\ ,\ a_3=\frac{13}{5}\ ,\ a_4=\frac{63}{17}\ ,\ a_5=\frac{62}{13}$$
\end{solbox}
\item $\ds .a_n=\frac{n^2}{2^n}$
\begin{solbox}
הסדרה נתונה על ידי
$a_n=\frac{n^2}{2^n}$
ולכן
$$.a_1=\frac{1^2}{2^1}=\frac12\ ,\ a_2=\frac{2^2}{2^2}=1\ ,\ a_3=\frac98\ ,\ a_4=1\ ,\ a_5=\frac{25}{32}$$
\end{solbox}
\item $\ds .a_n=1-\frac{(-1)^n}{n}$
\begin{solbox}
כאן
$a_n=1-\frac{(-1)^n}{n}$
ולכן
$$.a_1=1+\frac11=2\ ,\ a_2=1-\frac12=\frac12\ ,\ a_3=1+\frac13=\frac43\ ,\ a_4=1-\frac14=\frac34\ ,\ a_5=1+\frac15=\frac65$$
\end{solbox}
\eMs
\end{enumerate}
