\textbf{נתונה הסדרה
$a_n=\frac{n}{n^2+1}$.}
\begin{enumerate}
\item
הראו כי הסדרה מונוטונית.
\hint{הסתכלו על $a_{n+1}-a_n$.}
\begin{solbox}
    בשביל לבדוק מונוטוניות של הסדרה, נסתכל על ההפרש הבא
    \begin{align*}
        a_{n+1}-a_n=\frac{n+1}{(n+1)^2+1}-\frac{n}{n^2+1}
        =\frac{-n^2-n+1}{((n+1)^2+1)(n^2+1)}
    \end{align*}
    המכנה חיובי כמכפלה של שני חיוביים, והמונה שלילי לכל
    $n\ge 1$,
    כי
    מתקיים

    $-n^2-n+1\le -1-1+1=-1\le0$,
    לכן
    ההפרש שלילי, כלומר הסדרה מונוטונית יורדת (ממש).
\end{solbox}
\item
הראו כי הסדרה חסומה.
\hint{השתמשו בזהות $(a-b)^2=a^2-2ab+b^2\ge0$.}
\begin{solbox}
    נשתמש בזהות המוכרת ונקבל כי
    $$(n-1)^2=n^2-2n+1\ge0\Longrightarrow n^2+1\ge 2n\Longrightarrow \frac{1}{n^2+1}\le \frac{1}{2n}$$
    ואז כמובן
    $$a_n=\frac{n}{n^2+1}\le \frac{n}{2n}=\frac12$$
    כלומר הסדרה חסומה מלעיל על ידי
    $\frac12$,
    וכמובן שהיא חסומה מלרע על ידי
    $0$.
\end{solbox}
\end{enumerate}
