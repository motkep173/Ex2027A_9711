\textbf{הוכיחו שהסדרות הבאות מתכנסות ומצאו את גבולותיהן.}
\begin{enumerate}
\item $a_1=1$, $a_{n+1}=\sqrt{2+a_n}$.
\begin{solbox}
נכתוב כמה איברים ראשונים:
$$a_1=1,\qquad a_2=\sqrt3\approx1.73,\qquad a_3=\sqrt{2+\sqrt3}\approx1.93,\qquad a_4\approx1.98 .$$
לכן נחשוד שהסדרה עולה, ושאיבריה נשארים בין
$0$
ל-$2$.
נוכיח את שתי הטענות באינדוקציה.

\textbf{הסדרה עולה.}
נוכיח באינדוקציה ש-$a_{n+1}>a_n$ לכל $n$.
עבור $n=1$: $a_2=\sqrt3>1=a_1$.
נניח ש-$a_{n+1}>a_n$. אז $2+a_{n+1}>2+a_n$, ולכן
$$a_{n+2}=\sqrt{2+a_{n+1}}>\sqrt{2+a_n}=a_{n+1} .$$

\textbf{הסדרה חסומה: $0<a_n<2$ לכל $n$.}
עבור $n=1$: $0<1<2$.
נניח ש-$0<a_n<2$. אז $2<2+a_n<4$, ולכן
$$0<a_{n+1}=\sqrt{2+a_n}<\sqrt4=2 .$$

\textbf{הגבול.}
הסדרה עולה וחסומה מלעיל, ולכן מתכנסת. נסמן $a_n\to L$.
מכיוון ש-$a_n\ge a_1=1$ לכל $n$, גם $L\ge1$.
נעבור לגבול בשני האגפים של $a_{n+1}=\sqrt{2+a_n}$:
\begin{itemize}
\item באגף שמאל, $a_{n+1}$ היא אותה סדרה בהזזה של אינדקס, ולכן $a_{n+1}\goes L$.
\item באגף ימין, לפי אריתמטיקה של גבולות ושל חזקות, $\sqrt{2+a_n}=(2+a_n)^{1/2}\goes(2+L)^{1/2}$.
\end{itemize}
מיחידות הגבול, $L=\sqrt{2+L}$. נעלה בריבוע: $L^2-L-2=0$, כלומר $(L-2)(L+1)=0$.
מכיוון ש-$L\ge1$, נקבל $L=2$.
\end{solbox}
\item $a_1=3$, $\ds a_{n+1}=\frac12\lr{a_n+\frac{4}{a_n}}$.
\hint{כדי להראות שהסדרה חסומה מלרע, השתמשו באי-שוויון הממוצעים. כדי להראות שהיא מונוטונית, העריכו את ההפרש $a_{n+1}-a_n$.}
\begin{solbox}
\textbf{האיברים חיוביים.}
באינדוקציה: $a_1=3>0$, ואם $a_n>0$, אז גם $a_{n+1}=\frac12\lr{a_n+\frac4{a_n}}>0$ כסכום של חיוביים.

\textbf{הסדרה חסומה מלרע על ידי $2$.}
עבור $n=1$: $a_1=3\ge2$.
עבור $n\ge1$, האיבר $a_{n+1}$ הוא הממוצע החשבוני של שני המספרים החיוביים $a_n$ ו-$\frac4{a_n}$.
לפי אי-שוויון הממוצעים, הממוצע החשבוני גדול או שווה לממוצע ההנדסי:
$$a_{n+1}=\frac12\lr{a_n+\frac4{a_n}}\ \ge\ \sqrt{a_n\cdot\frac4{a_n}}=\sqrt4=2 .$$
לכן $a_n\ge2$ לכל $n$.

\textbf{הסדרה יורדת.}
נעריך את ההפרש:
$$a_{n+1}-a_n=\frac12\lr{a_n+\frac4{a_n}}-a_n=\frac{4}{2a_n}-\frac{a_n}{2}=\frac{4-a_n^2}{2a_n}.$$
המכנה $2a_n$ חיובי. המונה אי-חיובי, כי $a_n\ge2$ ולכן $a_n^2\ge4$.
לכן $a_{n+1}-a_n\le0$, כלומר $a_{n+1}\le a_n$ לכל $n$.

\textbf{הגבול.}
הסדרה יורדת וחסומה מלרע, ולכן מתכנסת. נסמן $a_n\to L$. מכיוון ש-$a_n\ge2$ לכל $n$, גם $L\ge2$, ובפרט $L\ne0$.
נעבור לגבול בשני האגפים של נוסחת הנסיגה:
\begin{itemize}
\item באגף שמאל, $a_{n+1}$ היא אותה סדרה בהזזה של אינדקס, ולכן $a_{n+1}\goes L$.
\item באגף ימין, לפי אריתמטיקה של גבולות (סכום, מנה כשגבול המכנה $L\ne0$, וכפל בקבוע), $\frac12\lr{a_n+\frac4{a_n}}\goes\frac12\lr{L+\frac4L}$.
\end{itemize}
מיחידות הגבול, $L=\frac12\lr{L+\frac4L}$. נכפול ב-$2L$: $2L^2=L^2+4$, כלומר $L^2=4$.
מכיוון ש-$L\ge2$, נקבל $L=2$.
\end{solbox}
\end{enumerate}
