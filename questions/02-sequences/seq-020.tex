\textbf{עבור כל אחת מהסדרות הבאות, קבעו אם היא חסומה, חסומה מלעיל ולא מלרע, חסומה מלרע ולא מלעיל, או לא חסומה מלעיל ולא מלרע. נמקו.}
\begin{enumerate}\bMs{2}
\item $\ds a_n=\frac{3n+(-1)^n}{n}$
\begin{solbox}
$a_n=3+\frac{(-1)^n}{n}$.
מכיוון ש-$-1\le\frac{(-1)^n}{n}\le\frac12$
(האיבר הגדול ביותר מתקבל ב-$n=2$),
מתקיים
$2\le a_n\le3.5$
לכל $n$.
לכן הסדרה חסומה.
\end{solbox}
\item $b_n=(-1)^n n$
\begin{solbox}
לא חסומה מלעיל ולא מלרע.
לכל $M$ יש $n$ זוגי כך ש-$n>M$, ואז $b_n=n>M$.
באותו אופן, לכל $m$ יש $n$ אי-זוגי כך ש-$n>-m$, ואז $b_n=-n<m$.
\end{solbox}
\item $\ds c_n=\frac{n^2+1}{n}$
\begin{solbox}
$c_n=n+\frac1n\ge n\ge1$,
ולכן הסדרה חסומה מלרע על ידי $1$.
היא לא חסומה מלעיל:
לכל $M$ יש $n>M$ טבעי (תכונת ארכימדס), ואז $c_n\ge n>M$.
\end{solbox}
\item $d_n=5-n^2$
\begin{solbox}
$d_n\le5-1=4$
לכל $n$, ולכן הסדרה חסומה מלעיל.
היא לא חסומה מלרע:
לכל $m$ יש $n$ טבעי כך ש-$n>5-m$, ואז
$d_n=5-n^2\le5-n<m$.
\end{solbox}
\item $\ds e_n=\frac{2n-1}{n+3}$
\begin{solbox}
$e_n>0$
לכל $n$, ו-$2n-1<2n+6=2(n+3)$,
ולכן
$0<e_n<2$.
הסדרה חסומה.

דרך נוספת: אם נשתמש במה שידוע לנו על התכנסות, נראה (יחסית בקלות) שהסדרה מתכנסת:
$$e_n=\frac{2-\frac1n}{1+\frac3n}\xrightarrow[n\to\infty]{}\frac{2-0}{1+0}=2,$$
לפי אריתמטיקה של גבולות. כל סדרה מתכנסת חסומה, ולכן גם מכאן נובע ש-$(e_n)$ חסומה.
\end{solbox}
\eMs
\end{enumerate}
