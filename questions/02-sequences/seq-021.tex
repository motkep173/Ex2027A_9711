הוכיחו לפי הגדרת הגבול שהסדרה
$a_n=(-1)^n+\frac1n$
אינה מתכנסת.
\begin{solbox}
עלינו להראות שאף מספר $L\in\rr$ אינו גבול של הסדרה, כלומר שלכל $L$ קיים
$\varepsilon>0$
כך שלכל $N$ קיים $n>N$ עם
$|a_n-L|\ge\varepsilon$.
ניקח
$\varepsilon=\frac12$
ונפריד לשני מקרים.

\textbf{מקרה 1: $L\ge0$.}
עבור $n$ אי-זוגי מתקיים
$a_n=-1+\frac1n$.
אם בנוסף $n\ge3$, אז $a_n\le-1+\frac13=-\frac23$, ולכן
$$|a_n-L|=L-a_n\ge0+\frac23>\frac12 .$$
לכל $N$ יש $n>N$ אי-זוגי עם $n\ge3$, ולכן $L$ אינו גבול.

\textbf{מקרה 2: $L<0$.}
עבור $n$ זוגי מתקיים
$a_n=1+\frac1n>1$, ולכן
$$|a_n-L|=a_n-L>1-0=1>\frac12 .$$
לכל $N$ יש $n>N$ זוגי, ולכן גם כאן $L$ אינו גבול.

בשני המקרים $L$ אינו גבול של הסדרה, ומכיוון ש-$L$ היה כלשהו, הסדרה אינה מתכנסת.
\end{solbox}
