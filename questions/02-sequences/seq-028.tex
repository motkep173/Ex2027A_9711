\textbf{לכל אחת מהסדרות הבאות, מצאו את הגבול
$\limn a_n$:}
\begin{enumerate}\bMs{2}
\item $\ds{\limn \dfrac{n^{700}}{3^n}}$

\hint{השתמשו במבחן המנה: חשבו את הגבול של $\frac{a_{n+1}}{a_n}$.}
\begin{solbox}
    נסמן
    $a_n=\dfrac{n^{700}}{3^n}$
    וננסה להשתמש במבחן המנה:
    מתקיים
    $$,\lim_{n\to\infty}\frac{a_{n+1}}{a_n}=\lim_{n\to\infty}\frac{\dfrac{(n+1)^{700}}{3^{n+1}}}{\dfrac{n^{700}}{3^n}}=\lim_{n\to\infty}\frac{1}{3}\left(1+\frac{1}{n}\right)^{700}=\frac{1}{3}$$
    מאחר ו-
    $\ds\lim_{n\to\infty}\left(1+\frac{1}{n}\right)=1$.
    מכיוון ש-
    $\frac{1}{3}<1$
    נסיק ממבחן המנה כי הגבול המבוקש הוא
    $0$.
הערה: בחישוב הגבול השתמשנו באריתמטיקה של גבולות: הגבול של מכפלה של סדרות מתכנסות שווה למכפלת הגבולות.
\end{solbox}

    %%%%

    \item $\ds{\limn \dfrac{100^n}{n!}}$

\hint{השתמשו במבחן המנה: חשבו את הגבול של $\frac{a_{n+1}}{a_n}$.}
\begin{solbox}
    נסמן
    $a_n=\dfrac{100^n}{n!}$.
    מתקיים
    $$,\lim_{n\to\infty}\frac{a_{n+1}}{a_n}=\lim_{n\to\infty}\frac{\dfrac{100^{n+1}}{(n+1)!}}{\dfrac{100^n}{n!}}=\lim_{n\to\infty}\frac{100}{n+1}=0$$ מכיוון ש-
    $0<1$
    נסיק, שוב ממבחן המנה, כי הגבול המבוקש הוא
    $0$.
\end{solbox}

    \item $\ds{\limn \dfrac{n!}{n^n}}$
\hint{השתמשו במבחן המנה. הגבול של $\frac{a_{n+1}}{a_n}$ קשור לגבול המגדיר את $e$.}
\begin{solbox}
          נסמן
    $a_n=\dfrac{n!}{n^n}$.
    מתקיים
    $$,\frac{a_{n+1}}{a_n}=
    \frac{\dfrac{(n+1)!}{(n+1)^{n+1}}}{\dfrac{n!}{n^n}}=
    \frac{n^n}{(n+1)^n}=\frac{1}{\left(1+\frac{1}{n}\right)^n}$$
והרי
$\ds\lim_{n\to\infty}\left(1+\frac{1}{n}\right)^n=e$,
    ולכן נקבל
    $$\limn \frac{a_{n+1}}{a_n}=\frac{1}{e}<1
    $$
    אז ממבחן המנה שוב
    נסיק כי הגבול המבוקש הוא
    $0$.
הערה: בחישוב הגבול השתמשנו באריתמטיקה של גבולות: הגבול של סכום, של מכפלה ושל מנה של סדרות מתכנסות שווה לסכום, למכפלה ולמנה של הגבולות, כל עוד הגבול של המכנה שונה מ-$0$.
\end{solbox}

\eMs
\end{enumerate}
