\textbf{מצאו נוסחאות לאיבר הכללי של הסדרות הבאות:}
\begin{enumerate}\bMs{2}
\item $.1,0,1,0,1,0\ldots$
\hint{התחילו מהסדרה $(-1)^n$ והפעילו עליה פעולות חשבון פשוטות.}
\begin{solbox}
נסמן את הסדרה ב-$(a_n)_{n=1}^{\infty}$.
ניזכר בסדרה הפשוטה
$$-1,1,-1,1,-1,1\ldots$$
שראינו בכיתה נתונה על ידי הנוסחה
$(-1)^n$.
על-ידי החסרת
$1$
מכל האיברים, כלומר בעזרת הנוסחה
$(-1)^n-1$
נקבל את הסדרה
$$-2,0,-2,0,-2,0,\ldots$$
ואז על-ידי חילוק כל האיברים ב-$(-2)$
נקבל את הסדרה הרצויה
$$.a_n=\frac{(-1)^n-1}{-2}$$
\end{solbox}
\item $\ds .\frac53,-\frac{10}{9},\frac{20}{27},-\frac{40}{81},\ldots$
\hint{שימו לב לסימן המתחלף, ובדקו פי כמה גדלים המונה והמכנה מאיבר לאיבר.}
\begin{solbox}
נשים לב כי איברי הסדרה הם
$$\frac53,\ (-1)\cdot\frac{5\cdot2}{3^2},\ \frac{5\cdot2^2}{3^3},\ (-1)\cdot\frac{5\cdot2^3}{3^4},\ldots$$
ולכן
$$.a_n=(-1)^{n+1}\cdot\frac{5\cdot2^{n-1}}{3^n}$$
\end{solbox}
\item $\ds .\frac{1}{1\cdot3},\frac{1}{3\cdot5},\frac{1}{5\cdot7},\ldots$
\begin{solbox}
מתקיים
$$.a_n=\frac{1}{(2n-1)(2n+1)}$$
\end{solbox}
\eMs
\end{enumerate}
