תהא $\ds f(x)=\ln\lr{1+2x^4}\cdot\sin\lr{3x^5}$. מצאו פולינום מקלורן מסדר $27$ של $f(x)$ וחשבו באמצעותו את $\ds f^{(27)}(0)$. (
את התשובה השאירו כמכפלה של מספרים.
)

\textbf{רמז:}
 חשבו על פולינומי מקלורן של
 $\ln(1+t)$
 ושל
 $\sin(t)$,
 מסדרים מסוימים.
\hint{פתחו כל גורם רק עד הסדר הדרוש כדי לקבל במכפלה את כל החזקות עד $x^{27}$, וזכרו שהמקדם של $x^{n}$ בפולינום מקלורן הוא $\frac{f^{(n)}(0)}{n!}$.}
\begin{solbox}
נזכר בפיתוח
של
$\ln(1+t)$:
$$,\ln(1+t)=t-\frac{t^2}{2}+\frac{t^3}{3}-\frac{t^4}{4}+\frac{t^5}{5}-\frac{t^6}{6}+o(t^6)$$
מכאן
\[
\ln(1+2x^4)=2x^4-\frac{(2x^4)^2}{2}+\frac{(2x^4)^3}{3}-\frac{(2x^4)^4}{4}
+\frac{(2x^4)^5}{5}-\frac{(2x^4)^6}{6}+o(x^{24}),
\]
כלומר
\[
.\ln(1+2x^4)=2x^4-2x^8+\frac{8}{3}x^{12}-4x^{16}+\frac{32}{5}x^{20}-\frac{32}{3}x^{24}+o(x^{25})
\]
בפיתוח של
$\sin t$
מסדר
$5$
נציב את
$t=3x^5$:
וכן
\[
.\sin(3x^5)=3x^5-\frac{(3x^5)^3}{6}+\frac{(3x^5)^5}{120}+O(x^{25})
=3x^5-\frac{9}{2}x^{15}+\frac{81}{40}x^{25}+o(x^{25})
\]
נכפיל ונשמור דרגות עד \(27\):
\[
.P_{27}(x)=6x^9-6x^{13}+8x^{17}-9x^{19}-12x^{21}+9x^{23}+\frac{96}{5}x^{25}-12x^{27}
\]
כפי שאנו רואים, המקדם של \(x^{27}\) הוא \(-12\), ולכן
מההגדרה של המקדמים בפיתוח מקלורן:
\[
,\frac{f^{(27)}(0)}{27!}=-12
\]
נסיק כי
$$.f^{(27)}(0)=-12\cdot 27!$$
\end{solbox}
