תהא $f(x)$ פונקציה רציפה בקטע \([a, b]\), גזירה בקטע \((a, b)\) ומקיימת
\\\(
\mbox{\(f^2(b) - f^2(a) = a^2 - b^2\)}
\).
הוכיחו כי למשוואה  \(\mbox{\(f(x) \cdot f'(x) + x = 0\)}\) יש לפחות פתרון אחד בקטע \((a, b)\).
\textbf{רמז:}  משפט רול.
\hint{חפשו פונקציית עזר שנגזרתה היא כפולה של $f(x)f'(x)+x$, וסדרו את הנתון כך שיראה שערכיה בקצות הקטע שווים.}
\begin{solbox}
נפשט
$$,f^2(a)+a^2=f^2(b)+b^2$$
ונסמן
$$.g(x)=f^2(x)+x^2$$
זוהי פונקציה רציפה בקטע
$[a,b]$
וגזירה בקטע
$(a,b)$
כסכום של פונקציות כאלה.
מתקיים
$$,g(a)-g(b)=f^2(a)+a^2-(f^2(b)+b^2)=0$$
כלומר
$g(a)=g(b)$.
ממשפט רול, קיים
$c\in(a,b)$
כך ש-
$$,0=g'(c)=2f(c)f'(c)+2c$$
הוכחנו עכשיו כי למשוואה
$$f(x)f'(x)+x=0$$
קיים פתרון
)$x=c$(.
\end{solbox}
