\textbf{
בכל אחד מהסעיפים הבאים,
חשבו את הגבול: }
\begin{enumerate}\bMs{2}
\item $\ds{\lim_{x\to\infty}\frac{2x^2-3x+6}{x^2-51x+1}}$
\ifsol
\begin{solbox}
מתקיים
\begin{align*}
    \frac{2x^2-3x+6}{x^2-51x+1}=\ \frac{x^2(2-\tfrac{3}{x}+\tfrac{6}{x^2})}{x^2(1-\tfrac{51}{x}+\tfrac{1}{x^2})}
    =\ \frac{2-\tfrac{3}{x}+\tfrac{6}{x^2}}{1-\tfrac{51}{x}+\tfrac{1}{x^2}},
\end{align*}
לכן
מאריתמטיקה של גבולות נקבל
\begin{align*}
    \ds{\lim_{x\to\infty}\frac{2x^2-3x+6}{x^2-51x+1}}
    =\ds{\lim_{x\to\infty}\frac{2-\tfrac{3}{x}+\tfrac{6}{x^2}}{1-\tfrac{51}{x}+\tfrac{1}{x^2}}}
    = \frac{2-0+0}{1-0+0}
    = 2,
\end{align*}
לפי אריתמטיקה של גבולות.
\end{solbox}
\fi

\item $\ds{ \lim_{x \to \infty} \lr{\frac{2x^3 - 4x^2 + 5}{x^2 + 2x +1}-2x}}$
\ifsol
\hint{אחדו את שני האיברים לשבר אחד לפני השאיפה לאינסוף.}
\begin{solbox}
מתקיים
\begin{align*}
    ,\lr{\frac{2x^3 - 4x^2 + 5}{x^2 + 2x +1}-2x}= \frac{ - 8x^2 -2x+ 5}{x^2 + 2x +1}
    = \frac{-8-\tfrac{2}{x}+\tfrac{5}{x^2}}{1+\frac{2}{x}+\frac{1}{x^2}}
\end{align*}
לכן
\begin{align*}
    .\ds{ \lim_{x \to \infty} \lr{\frac{2x^3 - 4x^2 + 5}{x^2 + 2x +1}-2x}}
    = \ds{ \lim_{x \to \infty}\frac{-8-\tfrac{2}{x}+\tfrac{5}{x^2}}{1+\frac{2}{x}+\frac{1}{x^2}}}
    = \frac{-8-0+0}{1+0+0}
    = -8
\end{align*}
הערה: בחישוב הגבול השתמשנו באריתמטיקה של גבולות: הגבול של סכום, של מכפלה ושל מנה של פונקציות מתכנסות שווה לסכום, למכפלה ולמנה של הגבולות, כל עוד הגבול של המכנה שונה מ-$0$.
\end{solbox}
\fi

\item $\ds{ \lim_{x \to \infty}\frac{x^3+3x-1}{\sqrt{9x^6+2}}}$
\ifsol
\begin{solbox}
מתקיים
\begin{align*}
    ,\frac{x^3+3x-1}{\sqrt{9x^6+2}}=\frac{x^3+3x-1}{\sqrt{x^6(9+\tfrac{2}{x^6})}}
    = \frac{x^3(1+\tfrac{3}{x^2}-\tfrac{1}{x^3})}{x^3\sqrt{9+\tfrac{2}{x^6}}}
    = \frac{1+\tfrac{3}{x^2}-\tfrac{1}{x^3}}{\sqrt{9+\tfrac{2}{x^6}}}
\end{align*}
לכן
\begin{align*}
    \ds{ \lim_{x \to \infty}\frac{x^3+3x-1}{\sqrt{9x^6+2}}}= \ds{ \lim_{x \to \infty}\frac{1+\tfrac{3}{x^2}-\tfrac{1}{x^3}}{\sqrt{9+\tfrac{2}{x^6}}}}
    = \frac{1}{3},
\end{align*}
הערה: בחישוב הגבול השתמשנו באריתמטיקה של גבולות: הגבול של סכום, של מכפלה ושל מנה של פונקציות מתכנסות שווה לסכום, למכפלה ולמנה של הגבולות, כל עוד הגבול של המכנה שונה מ-$0$ בנוסף, השורש הוא פונקציה רציפה, ולכן אפשר להעביר את הגבול אל תוך השורש.
\end{solbox}
\fi

\item $\ds{\lim_{x\to\infty}\frac{\cos(x)}{x}}$
\ifsol
\hint{מה ידוע על מכפלה של פונקציה חסומה בפונקציה ששואפת לאפס?}
\begin{solbox}
נשים לב כי
$|\cos(x)|\leq 1$
לכל
$x\in\mathbb{R}$,
כלומר,
$\cos(x)$
היא פונקציה חסומה. בנוסף
$\displaystyle\lim_{x\to\infty}\frac{1}{x}=0$.
נסיק כי
$\ds{\lim_{x\to\infty}\frac{\cos(x)}{x}}=0$
כמכפלה של פונקציה חסומה באפסה.
\end{solbox}
\fi

\item $\ds{\lim_{x\to\infty}\sin(x)\cdot\sin\left(\tfrac{1}{x}\right)}$
\ifsol
\hint{מה ידוע על מכפלה של פונקציה חסומה בפונקציה ששואפת לאפס?}
\begin{solbox}
נשים לב כי
$|\sin(x)|\leq 1$
לכל
$x\in\mathbb{R}$,
כלומר,
$\sin(x)$
היא פונקציה חסומה. בנוסף
$\displaystyle\lim_{x\to\infty}\sin\left(\frac{1}{x}\right)=0$
נסיק כי
$\ds{\lim_{x\to\infty}\sin(x)\cdot\sin\left(\tfrac{1}{x}\right)}=0$.
כמכפלה של פונקציה חסומה באפסה.
\end{solbox}
\fi

\item $\ds{ \lim_{x \to -\infty} \frac{x}{\sqrt{2x^2 + 1}}}$
\ifsol
\hint{הוציאו את $x^2$ מתוך השורש ושימו לב ש-$\sqrt{x^2}=|x|$. מה הסימן של $x$ כאן?}
\begin{solbox}
מתקיים
\begin{align*}
    ,\frac{x}{\sqrt{2x^2 + 1}}=
    \frac{x}{|x|\sqrt{2 + \frac{1}{x^2}}}
    =
    \frac{x}{-x\sqrt{2 + \frac{1}{x^2}}}
    \overset{(\ast)}{=}
    -\frac{1}{\sqrt{2 + \frac{1}{x^2}}},
\end{align*}
לכן
\begin{align*}
    .\lim_{x \to -\infty} \frac{x}{\sqrt{2x^2 + 1}}
    =
    -\lim_{x \to -\infty} \frac{1}{\sqrt{2 + \frac{1}{x^2}}}
    =\ -\frac{1}{\sqrt{2 + 0}}
    = -\frac{1}{\sqrt{2}}
\end{align*}
$(\ast)$
    יש כאן שאיפה למינוס אינסוף, לכן
    $|x|=-x$.
הערה: בחישוב הגבול השתמשנו באריתמטיקה של גבולות: הגבול של סכום, של מכפלה ושל מנה של פונקציות מתכנסות שווה לסכום, למכפלה ולמנה של הגבולות, כל עוד הגבול של המכנה שונה מ-$0$ בנוסף, השורש הוא פונקציה רציפה, ולכן אפשר להעביר את הגבול אל תוך השורש.
\end{solbox}
\fi

\item $\ds{ \lim_{x \to \infty}(\sqrt{x^2+1}-\sqrt{x^2-1})}$
\ifsol
\hint{זהו ביטוי מהצורה $\infty-\infty$ עם שורשים: הכפילו וחלקו בצמוד.}
\begin{solbox}
מתקיים
\begin{align*}
    (\sqrt{x^2+1}-\sqrt{x^2-1})&=\ (\sqrt{x^2+1}-\sqrt{x^2-1})\cdot{\color{red}\frac{\sqrt{x^2+1}+\sqrt{x^2-1}}{\sqrt{x^2+1}+\sqrt{x^2-1}}}\\
    &=\ \frac{2}{\sqrt{x^2+1}+\sqrt{x^2-1}}
\end{align*}
לכן
\begin{align*}
    \lim_{x \to \infty}(\sqrt{x^2+1}-\sqrt{x^2-1})
    &=\ \lim_{x \to \infty}\frac{2}{\sqrt{x^2+1}+\sqrt{x^2-1}}\\
    &=\ 0.
\end{align*}
\end{solbox}
\fi

\item $\ds{ \lim_{x \to \infty} (2x+3)\lr{\ln(x + 2) - \ln(x)}}$

\ifsol
\hint{השתמשו בחוקי הלוגריתמים כדי לכתוב את הביטוי כ-$\ln$ של חזקה, ושימו לב שבתוך ה-$\ln$ יש גבול מהצורה $1^\infty$.}
\begin{solbox}
ניעזר בחוקי לוגריתמים ונקבל
\begin{align*}
    (2x+3)\lr{\ln(x + 2) - \ln(x)}=(2x+3)\ln(1+\tfrac{2}{x})
    = \ln\left((1+\tfrac{2}{x})^{2x+3}\right)
    \end{align*}
    הביטוי בתוך ה-$\ln$ דומה לגבול המופלא המוכר, ולכן נמשיך לפשט רק אותו  ה-$\ln$-
    \begin{align*}
    (1+\tfrac{2}{x})^{2x+3}
    &=\ (1+\tfrac{2}{x})^{{\color{red}\frac{x}{2}}\cdot(2x+3)\cdot{\color{red}\frac{2}{x}}}\\
    &=\ (1+\tfrac{2}{x})^{{\frac{x}{2}}\cdot{(4+\tfrac{6}{x})}}
\end{align*}
לכן
\begin{align*}
    \lim_{x\to\infty}(2x+3)\lr{\ln(x + 2) - \ln(x)}&=\ \lim_{x\to\infty}(4+\tfrac{6}{x})\ln(1+\tfrac{2}{x})^{\frac{x}{2}}\\
    &=\ 4\cdot\ln e
    = 4.
\end{align*}
הערה: בחישוב הגבול השתמשנו באריתמטיקה של גבולות: הגבול של מכפלה של פונקציות מתכנסות שווה למכפלת הגבולות.
\end{solbox}
\fi

\item $\ds\lim_{x\to\infty}\lr{\frac{x+2}{x-1}}^{x}$
\ifsol
\begin{solbox}
נכתוב $\frac{x+2}{x-1}=1+\frac{3}{x-1}$, ולכן
\[\lr{\frac{x+2}{x-1}}^{x}=\lr{\lr{1+\frac{3}{x-1}}^{\frac{x-1}{3}}}^{\frac{3x}{x-1}}\]
כאשר $x\to\infty$ מתקיים $\frac{3}{x-1}\to0$, ולכן לפי הגבול המופלא הבסיס החיצוני שואף ל-$e$. בנוסף $\ds\lim_{x\to\infty}\frac{3x}{x-1}=3$. לפי גבול של חזקה:
\[\lim_{x\to\infty}\lr{\frac{x+2}{x-1}}^{x}=e^3\]
\end{solbox}
\fi

\item $\ds\lim_{x\to\infty}\frac{3^{x+1}+2^{x}}{3^{x}-2^{x+2}}$
\ifsol
\hint{חלקו את המונה ואת המכנה בגורם הדומיננטי $3^x$.}
\begin{solbox}
\unverified
הביטוי מהצורה $\frac{\infty}{\infty}$ (כי $\ds\lim_{x\to\infty}3^x=\infty$). נחלק את המונה ואת המכנה ב-$3^x>0$, ונשתמש ב-$3^{x+1}=3\cdot3^x$ וב-$2^{x+2}=4\cdot2^x$:
\[\frac{3^{x+1}+2^{x}}{3^{x}-2^{x+2}}=\frac{3+\lr{\frac23}^x}{1-4\lr{\frac23}^x}.\]
מאריתמטיקה של חזקות (הבסיס $\frac23$ הוא קבוע בין $0$ ל-$1$, והמעריך שואף ל-$\infty$) נובע ש-$\ds\lim_{x\to\infty}\lr{\frac23}^x=0$.
בפרט המכנה שואף ל-$1-4\cdot0=1\ne0$, ולכן הוא שונה מ-$0$ לכל $x$ גדול מספיק. לפי אריתמטיקה של גבולות:
\[\lim_{x\to\infty}\frac{3^{x+1}+2^{x}}{3^{x}-2^{x+2}}=\frac{3+0}{1-0}=3.\]
\end{solbox}
\fi

\item $\ds\lim_{x\to\infty}\lr{1+\sin\lr{\frac{2}{x}}}^{x}$
\ifsol
\hint{צורה $1^\infty$ עם $u=\sin\lr{\frac2x}$. כתבו את המעריך כ-$\frac1u\cdot ux$ וחשבו את $\lim ux$ בעזרת $\frac{\sin(t)}{t}\to1$.}
\begin{solbox}
נסמן $u=\sin\lr{\frac2x}$. עבור $x>\frac4\pi$ מתקיים $0<\frac2x<\frac\pi2$, ולכן $0<u<1$; בפרט $u\ne0$ והבסיס חיובי. כאשר $x\to\infty$ מתקיים $\frac2x\to0$, ולכן (הצבה $t=\frac2x$ וגבול $\sin(t)\to\sin(0)=0$) $u\to0$. זו צורה $1^\infty$, ונכתוב:
\[\lr{1+\sin\lr{\frac{2}{x}}}^{x}=\left[(1+u)^{\frac1u}\right]^{ux}.\]
\begin{itemize}
\item הבסיס: לפי הגבול המופלא $(1+u)^{\frac1u}\to e$ וגבול של הרכבה ($u\to0$ ו-$u\ne0$), $(1+u)^{\frac1u}\to e$.
\item המעריך: נציב $t=\frac2x$; כאשר $x\to\infty$ מתקיים $t\to0^+$, ו-$t\ne0$. לכן
\[\lim_{x\to\infty}ux=\lim_{x\to\infty}x\sin\lr{\frac2x}=\lim_{t\to0^+}2\cdot\frac{\sin(t)}{t}=2\cdot1=2\]
לפי הגבול המופלא $\frac{\sin(t)}{t}\to1$ ואריתמטיקה של גבולות.
\end{itemize}
לפי כלל החזקה באריתמטיקה של גבולות (הבסיס שואף ל-$e>0$ והמעריך ל-$2$):
\[\lim_{x\to\infty}\lr{1+\sin\lr{\frac{2}{x}}}^{x}=e^2.\]
\end{solbox}
\fi

\item $\ds\lim_{x\to\infty}\frac{\floor{x^2}}{x^2+x}$
\ifsol
\hint{השתמשו ב-$t-1<\floor{t}\le t$ עם $t=x^2$, ובמשפט הסנדוויץ'.}
\begin{solbox}
לכל $t\in\rr$ מתקיים $t-1<\floor t\le t$. עם $t=x^2$:
\[x^2-1<\floor{x^2}\le x^2.\]
לכל $x>0$ מתקיים $x^2+x>0$, ולכן מותר לחלק בו בלי לשנות את כיוון אי-השוויונות:
\[\frac{x^2-1}{x^2+x}<\frac{\floor{x^2}}{x^2+x}\le\frac{x^2}{x^2+x}.\]
נחשב את הגבולות של החסמים. בחלוקת המונה והמכנה ב-$x^2$ ולפי $\frac{c}{\pm\infty}=0$ ואריתמטיקה של גבולות:
\[\lim_{x\to\infty}\frac{x^2-1}{x^2+x}=\lim_{x\to\infty}\frac{1-\frac{1}{x^2}}{1+\frac1x}=\frac{1-0}{1+0}=1\]
\[\lim_{x\to\infty}\frac{x^2}{x^2+x}=\lim_{x\to\infty}\frac{1}{1+\frac1x}=1\]
לפי משפט הסנדוויץ':
\[\lim_{x\to\infty}\frac{\floor{x^2}}{x^2+x}=1.\]
\end{solbox}
\fi

\item $\ds\lim_{x\to\infty}\frac{x\sin(x)+2\cos\lr{x^2}}{x^2+1}$
\ifsol
\hint{חלקו את המונה ואת המכנה בגורם הדומיננטי $x^2$. מה מקבלים במונה?}
\begin{solbox}
\unverified
הגבול $\ds\lim_{x\to\infty}\sin(x)$ אינו קיים, ולכן אי אפשר להשתמש בכלל הכפל. נחלק את המונה ואת המכנה בגורם הדומיננטי $x^2$:
\[\frac{x\sin(x)+2\cos\lr{x^2}}{x^2+1}=\frac{\frac1x\cdot\sin(x)+\frac{2}{x^2}\cdot\cos\lr{x^2}}{1+\frac{1}{x^2}}\]
במונה, $\ds\lim_{x\to\infty}\frac1x=0$ ו-$\abs{\sin(x)}\le1$, וגם $\ds\lim_{x\to\infty}\frac{2}{x^2}=0$ ו-$\abs{\cos\lr{x^2}}\le1$. לכן לפי ``אפסה כפול חסומה'' שני המחוברים במונה שואפים ל-$0$, והמונה שואף ל-$0$. המכנה שואף ל-$1+0=1$. לפי אריתמטיקה של גבולות:
\[\lim_{x\to\infty}\frac{x\sin(x)+2\cos\lr{x^2}}{x^2+1}=\frac{0}{1}=0\]
\end{solbox}
\fi

\eMs
\end{enumerate}
