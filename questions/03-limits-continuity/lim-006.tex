מצאו את כל הערכים של הפרמטרים
$a,b$
עבורם קיים
${\displaystyle\lim_{x\to0}}f(x)$,
כאשר
\[
.f(x) = \case{cc}{
       2x+a, &  x\leq0\\ \ln(b+12x^2), & x>0}
\]
\ifsol
\hint{חשבו את שני הגבולות החד-צדדיים ודרשו שיהיו שווים. שימו לב גם לתחום ההגדרה של $\ln$.}
\begin{solbox}
מתקיים
\begin{align*}
    \lim_{x\to0^-}f(x)= \lim_{x\to0^-}(2x+a)
    = 2\cdot0+a
    = a,
\end{align*}
ו-
\begin{align*}
    \lim_{x\to0^+}f(x)= \lim_{x\to0^+}\ln(b+12x^2)
    = \ln(b+12\cdot0^2)
    =\ln(b),
\end{align*}
לכן
$\lim_{x\to0}f(x)$
קיים אמ"מ
$\lim_{x\to0^+}f(x)=\lim_{x\to0^-}f(x)$,
כלומר
$a=\ln(b)$
ו-
$b>0$.
\end{solbox}
\fi
