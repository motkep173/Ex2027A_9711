מצאו את כל הערכים של $a$ עבורם $\ds\lim_{x\to0}\frac{\sin(ax)}{\sqrt{1+x}-1}=6$.
\ifsol
\hint{כפלו וחלקו בצמוד של המכנה, והשלימו לגבול המופלא של $\frac{\sin t}{t}$.}
\begin{solbox}
אם $a=0$, הביטוי שווה ל-$0$ לכל $x\ne0$, והגבול הוא $0\ne6$. נניח $a\ne0$. לכל $0<|x|<1$ נכפול ונחלק בצמוד $\sqrt{1+x}+1$:
\[\frac{\sin(ax)}{\sqrt{1+x}-1}=\frac{\sin(ax)}{x}\lr{\sqrt{1+x}+1}=a\cdot\frac{\sin(ax)}{ax}\cdot\lr{\sqrt{1+x}+1}\]
כאשר $x\to0$ מתקיים $ax\to0$, ו-$ax\ne0$ לכל $x\ne0$. לפי הגבול המופלא $\ds\lim_{t\to0}\frac{\sin t}{t}=1$ ואריתמטיקה של גבולות:
\[\lim_{x\to0}\frac{\sin(ax)}{\sqrt{1+x}-1}=a\cdot1\cdot(1+1)=2a\]
לכן $2a=6$, כלומר $a=3$.
\end{solbox}
\fi
