נתון כי
$\ds{\lim_{x \to 3} f(x)  =  7}$. הוכיחו כי קיימת סביבה נקובה של
$ \ds{x  =  3} $
אשר בה
$ \ds{f(x)  >  6} $.
\ifsol
\hint{הציבו בהגדרת הגבול $\varepsilon$ מתאים, כך שהקרבה של $f(x)$ ל-$7$ תבטיח את אי-השוויון.}
\begin{solbox}
נקבע
$\varepsilon=1$.
מהגדרת הגבול של פונקציה, עבור האפסילון שקבענו זה עתה, קיים
$\delta>0$
כך ש-
$|f(x)-7|<1$
כל עוד
$x\in(3-\delta,3+\delta)$,
כלומר,
$$,6<f(x)<8,\ \forall x\in(3-\delta,3+\delta)$$
בפרט
$f(x)>6$,
כמבוקש.
\end{solbox}
\fi
