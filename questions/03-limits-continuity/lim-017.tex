נתונה
$f$
פונקציה רציפה על כל הישר המקיימת
$\ds{\abs{f(x)}\leq 5}$
לכל
$x\in\rr$.

הוכיחו כי קיים
$c\in\mathbb{R}$
עבורו
$f(c)=9-3c$.
\hint{הגדירו פונקציית עזר $g(x)=f(x)-(9-3x)$ ומצאו, בעזרת החסם על $f$, נקודות שבהן הסימן של $g$ ידוע.}
\begin{solbox}
נסמן
$g(x)=3x+f(x)-9$.
מתקיים
$g$
רציפה לכל
$x\in\mathbb{R}$.
בנוסף
$$,g(6)=18+f(6)-9=9+f(6)\geq9-5=4>0$$
כמוכן
$$.g(0)=f(0)-9\leq 5-9=-4<0$$
ממשפט ערך הביניים, קיים
$c\in(0,6)$
כך ש-
$g(c)=0$,
כלומר,
$$,3c+f(c)=9$$
ולכן
$$.f(c)=9-3c$$
כנדרש.
\end{solbox}
