\textbf{חשבו את הגבולות הבאים, או הראו שהגבול אינו קיים.}
\begin{enumerate}
\item $\ds\lim_{x\to1}\lr{\floor{x}+\floor{-x}}$
\ifsol
\hint{חשבו בנפרד את הגבול מימין ואת הגבול משמאל. מה הם $\floor{x}$ ו-$\floor{-x}$ כאשר $1<x<2$, וכאשר $0<x<1$?}
\begin{solbox}
נחשב את שני הגבולות החד-צדדיים.
\begin{itemize}
\item אם $1<x<2$, אז $\floor x=1$, ו-$-2<-x<-1$ ולכן $\floor{-x}=-2$. בקטע הזה $\floor{x}+\floor{-x}=-1$, ולכן $\ds\lim_{x\to1^+}\lr{\floor{x}+\floor{-x}}=-1$.
\item אם $0<x<1$, אז $\floor x=0$, ו-$-1<-x<0$ ולכן $\floor{-x}=-1$. בקטע הזה $\floor{x}+\floor{-x}=-1$, ולכן $\ds\lim_{x\to1^-}\lr{\floor{x}+\floor{-x}}=-1$.
\end{itemize}
שני הגבולות החד-צדדיים קיימים ושווים, ולכן לפי המשפט על גבולות חד-צדדיים
\[\lim_{x\to1}\lr{\floor{x}+\floor{-x}}=-1.\]
שימו לב: ערך הפונקציה בנקודה עצמה הוא $\floor1+\floor{-1}=0\ne-1$, אבל הגבול אינו תלוי בערך הזה. אסור לחשב כאן ``$\lim\floor x+\lim\floor{-x}$'', כי לאף אחד מהמחוברים אין גבול ב-$1$.
\end{solbox}
\fi
\end{enumerate}
