\textbf{הוכיחו או הפריכו כל אחד מהסעיפים הבאים:}
\begin{enumerate}
\item אם $\ds\lim_{x\to x_0}f(x)=L$ ו-$L>0$, אז קיימת סביבה נקובה של $x_0$ שבה $f(x)>0$.
\ifsol
\hint{נסו לשרטט את הנתונים, ואז נסו לבחור $\varepsilon$ מתאים.}
\begin{solbox}
הטענה נכונה. לפי הגדרת הגבול עם $\varepsilon=\frac L2>0$, קיים $\delta>0$ כך שלכל $0<|x-x_0|<\delta$ מתקיים $|f(x)-L|<\frac L2$, ובפרט
\[f(x)>L-\frac L2=\frac L2>0\]
באיור: בסביבה הנקובה $(x_0-\delta,x_0)\cup(x_0,x_0+\delta)$ הגרף נמצא בתוך הרצועה שבין $\frac L2$ ל-$\frac{3L}{2}$, ולכן מעל ציר ה-$x$.
\qfig{lim-033a}
\end{solbox}
\fi
\item אם $f(x)>0$ לכל $x\in\rr$, וקיים הגבול $\ds\lim_{x\to x_0}f(x)=L$, אז $L>0$.
\ifsol
\begin{solbox}
הטענה אינה נכונה. נגדיר
\[f(x)=\begin{cases}x^2, & x\ne0\\ 1, & x=0\end{cases}\]
לכל $x\in\rr$ מתקיים $f(x)>0$, אבל $\ds\lim_{x\to0}f(x)=\lim_{x\to0}x^2=0$. (אפשר להוכיח רק ש-$L\ge0$.)
\end{solbox}
\fi
\item אם $\ds\lim_{x\to x_0}f(x)=L$, אז $f(x_0)=L$.
\ifsol
\begin{solbox}
הטענה אינה נכונה. ראשית, $f$ לא חייבת להיות מוגדרת ב-$x_0$: למשל $f(x)=\frac{x^2}{x}$ מוגדרת לכל $x\ne0$, ומתקיים $\ds\lim_{x\to0}f(x)=\lim_{x\to0}x=0$, אבל $f(0)$ אינו מוגדר. שנית, גם כש-$f$ מוגדרת ב-$x_0$, הערך יכול להיות שונה מהגבול. למשל
\[f(x)=\begin{cases}0, & x\ne0\\ 1, & x=0\end{cases}\]
כאן $\ds\lim_{x\to0}f(x)=0\ne1=f(0)$, כי הגבול אינו תלוי בערך בנקודה עצמה.
\end{solbox}
\fi
\item אם $f$ חסומה בסביבה נקובה של $x_0$, אז קיים הגבול $\ds\lim_{x\to x_0}f(x)$.
\ifsol
\begin{solbox}
הטענה אינה נכונה. למשל $f(x)=\floor{x}$ ו-$x_0=1$. בקטע $(0,2)$ מתקיים $0\le f(x)\le1$, ולכן $f$ חסומה שם. נניח בשלילה ש-$\ds\lim_{x\to1}f(x)=L$, וניקח $\varepsilon=\frac12$. אז קיים $\delta>0$ שעבורו בקטע $(1-\delta,1+\delta)$, מלבד $1$, יש גם נקודות שבהן $f=0$ וגם נקודות שבהן $f=1$. לכן
\[1=|1-0|\le|1-L|+|L-0|<\frac12+\frac12=1\]
סתירה.
\end{solbox}
\fi
\end{enumerate}
