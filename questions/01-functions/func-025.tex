\textbf{בכל אחד מהזוגות הבאים, חשבו את
$A\cup B$, $A\cap B$, $A\setminus B$, $B\setminus A$:}
\begin{enumerate}\bMs{2}
\item $.B=[0,3)\ ,\ A=(-1,2)$
\begin{solbox}
מתקיים
$$.A\cup B=(-1,3),\ A\cap B=[0,2),\ A\setminus B=(-1,0),\ B\setminus A=[2,3)$$
\end{solbox}
\item $.B=(1,5)\ ,\ A=(-\infty,3]$
\begin{solbox}
מתקיים
$$.A\cup B=(-\infty,5),\ A\cap B=(1,3],\ A\setminus B=(-\infty,1],\ B\setminus A=(3,5)$$
\end{solbox}
\item $.B=(1,\infty)\ ,\ A=(-\infty,2]$
\begin{solbox}
מתקיים
$$.A\cup B=\mathbb{R},\ A\cap B=(1,2],\ A\setminus B=(-\infty,1],\ B\setminus A=(2,\infty)$$
\end{solbox}
\eMs
\end{enumerate}
