תהי $f$ פונקציה שהיא גם זוגית וגם אי-זוגית. הוכיחו כי $f(x)=0$ לכל $x$ בתחום ההגדרה שלה.
\hint{כתבו את $f(-x)$ בשתי דרכים: פעם לפי הזוגיות ופעם לפי האי-זוגיות.}
\begin{solbox}
$f(-x)=f(x)$
מכך שזוגית
$f(-x)=-f(x)$
מכך שאי זוגית ולכן
\\
$-f(x)=f(-x)=f(x)$
לכן
$f(x)=0$
.
\end{solbox}
