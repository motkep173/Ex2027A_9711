הציגו את פונקציה $f(x)=2^x$
כסכום של פונקציה זוגית $g(x)$
ופונקציה אי זוגית $h(x)$
.
\hint{השתמשו בנוסחאות לחלק הזוגי ולחלק האי-זוגי של פונקציה מהשאלה על פירוק פונקציה לסכום של זוגית ואי-זוגית.}
\begin{solbox}
בהסתמך על שאלה \qref{func-021} נקבל
$g(x)=\frac{f(x)+f(-x)}{2}=\frac{2^x+2^{-x}}{2}$
\\
$h(x)=\frac{f(x)-f(-x)}{2}=\frac{2^x-2^{-x}}{2}$
\end{solbox}
