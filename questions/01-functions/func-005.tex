\textbf{מצאו את תחום ההגדרה הטבעי של הפונקציות הבאות:}
\begin{enumerate}\bMs{2}
\item $f(x)=\sqrt{x-1}$
\begin{solbox}
תחום ההגדרה הוא $[1,\infty)$.
\end{solbox}

\item $f(x)=\sqrt[6]{x^2+1}$
\begin{solbox}
תחום ההגדרה הוא $\mathbb{R}$, כי $x^2+1>0$ לכל $x$.
\end{solbox}

\item  $f(x)=\ds{\frac{\sin x}{\sqrt{1-x^2}}}$
\begin{solbox}
תחום ההגדרה הוא $(-1,1)$, כי צריך שיתקיים
$1-x^2>0$.
\end{solbox}

\item $f(x)=\frac{x}{\sin x}$
\begin{solbox}
תחום ההגדרה הוא $\mathbb{R}\setminus\{k\pi:k\in\mathbb{Z}\}$,
כי צריך שיתקיים
$\sin x\neq0$.
\end{solbox}

\item $f(x)=\log_2{(x^2-5x+6)}$
\begin{solbox}
צריך שיתקיים $x^2-5x+6>0$, כלומר $(x-2)(x-3)>0$, ולכן התחום הוא $(-\infty,2)\cup(3,\infty)$.
\end{solbox}

\item $f(x)=\tan(\pi x)$
\begin{solbox}
צריך שיתקיים $\cos(\pi x)\neq0$, ולכן $x\neq k+\frac12$, $k\in\mathbb{Z}$. התחום הוא $\mathbb{R}\setminus\{k+\frac12:k\in\mathbb{Z}\}$.
\end{solbox}

\item $f(x)=\ds{\sqrt{\frac{x+3}{\abs{x^2-4}}}}$
\hint{שימו לב שהמכנה אי-שלילי, ולכן סימן השבר נקבע לפי המונה. אל תשכחו לדרוש שהמכנה לא יתאפס.}
\begin{solbox}
צריך שיתקיים $x\neq\pm2$ וגם $\frac{x+3}{|x^2-4|}\geq0$. מאחר שהמכנה חיובי בכל מקום שבו הוא מוגדר, נקבל $x\geq-3$. לכן התחום הוא
$[-3,-2)\cup(-2,2)\cup(2,\infty)$.
\end{solbox}

\item  $f(x) = (-x)^{\frac{3}{2}}$
\begin{solbox}
$(- \infty, 0]$
\end{solbox}

\item  $f(x) = \sqrt{\sin(\sqrt{x})}$
\hint{דרשו ש-$x\geq0$ וש-$\sin(\sqrt x)\geq0$. סמנו $t=\sqrt x$ ושאלו את עצמכם בצד: מתי $\sin t$ אי-שלילי?}
\begin{solbox}
כדי ש-$\sqrt x$ יהיה מוגדר צריך שיתקיים $x\geq0$, וכדי שהשורש החיצוני יהיה מוגדר צריך שיתקיים
$\sin(\sqrt x)\geq0$.

נסמן $t=\sqrt x$, כך ש-$t\geq0$, ונבדוק מתי $\sin t\geq0$. הפונקציה $\sin$ אי-שלילית בקטע $[0,\pi]$ ושלילית בקטע $(\pi,2\pi)$, והיא מחזורית עם מחזור $2\pi$. לכן עבור $t\geq0$ מתקיים $\sin t\geq 0$ בדיוק כאשר
$$2k\pi\leq t\leq (2k+1)\pi$$
עבור $k=0,1,2,\ldots$ כלשהו.

נחזור ל-$x$. שלושת המספרים באי-השוויון
$2k\pi\leq \sqrt x\leq (2k+1)\pi$
אי-שליליים, ולכן הוא שקול לאי-השוויון שמתקבל כשמעלים בריבוע:
$$.4k^2\pi^2\leq x\leq (2k+1)^2\pi^2$$
לכן תחום ההגדרה הוא
$$.\bigcup_{k=0}^{\infty} \left[4k^2 \pi^2, (2k + 1)^2 \pi^2 \right]=[0,\pi^2]\cup[4\pi^2,9\pi^2]\cup[16\pi^2,25\pi^2]\cup\cdots$$
\end{solbox}

\item  $f(x) = \sqrt[4]{\log_2(\tan x)}$
\hint{תחום ההגדרה של הלוגריתם הוא המספרים החיוביים. אחרי שכתבתם את הדרישות, היעזרו במחזוריות של $\tan$: מספיק לפתור בקטע $(-\frac{\pi}{2},\frac{\pi}{2})$.}
\begin{solbox}
ממחזוריות של
$\tan$
נוכל לפתור את השאלה בקטע
$(-\frac{\pi}{2}, \frac{\pi}{2})$
ואז לשכפל את התוצאה על הישר כולו.
כדי שפונקציית הלוג תוגדר, היא צריכה לקבל ערכים חיוביים. לכן נדרוש
$\tan x > 0$.
זה קורה בקטע
$(0, \frac{\pi}{2})$.
כדי שנוכל להוציא שורש רביעי ממספר הוא צריך להיות אי שלילי. לכן נדרוש
$\log_2 (\tan x) \geq 0$.
זה קורה כאשר
$\tan x \geq 1$,
וזה קורה בקטע
$[\frac{\pi}{4}, \frac{\pi}{2})$.
כעת נשכפל את מה שקיבלנו לישר, ונקבל שתחום ההגדרה הוא
  $\bigcup_{k=-\infty}^{\infty} [\frac{\pi}{4} + k \pi, \frac{\pi}{2} + k \pi )$
\end{solbox}
\eMs
\end{enumerate}
