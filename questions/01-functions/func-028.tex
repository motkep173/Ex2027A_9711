\begin{enumerate}
\item
מצאו
$n\in\mathbb{N}$
כך ש-$\frac1n<0.003$.
\begin{solbox}
$\frac1n<0.003$
אם ורק אם
$n>\frac{1}{0.003}=333.\overline{3}$.
לכן למשל
$n=334$
מתאים (וכך גם כל $n$ גדול יותר).
\end{solbox}
\item
מצאו
$n\in\mathbb{N}$
כך ש-$\frac{1}{\sqrt n}<0.01$.
\begin{solbox}
$\frac{1}{\sqrt n}<0.01$
אם ורק אם
$\sqrt n>100$,
כלומר
$n>10000$.
לכן למשל
$n=10001$
מתאים.
\end{solbox}
\item
מצאו מספר רציונלי
$q$
כך ש-$\sqrt2<q<1.415$.
בתרגיל זה מותר להשתמש במחשבון כדי להעלות מספר בריבוע, אבל לא כדי להוציא שורש.
\hint{בחרו מספר עשרוני סופי קטן מ-$1.415$, והראו שהוא גדול מ-$\sqrt2$ בעזרת העלאה בריבוע.}
\begin{solbox}
ניקח
$q=1.4145=\frac{14145}{10000}\in\mathbb{Q}$.
ברור ש-$q<1.415$.
בנוסף,
$q^2=2.00081025>2$.
שני האגפים חיוביים, ולכן
$q>\sqrt2$.
\end{solbox}
\end{enumerate}
