\textbf{מצאו את ההרכבות
$f(g(x)),g(f(x)),f(f(x)),g(g(x))$:}
\begin{enumerate}\bMs{2}
  %\item {$f: \mathbb{R} \rightarrow \mathbb{R},\f(x)=x^3$}
 %\item{$f: \mathbb{R} \rightarrow \mathbb{R}, f(x)=\sin(x)$}
 %\item{$f: \mathbb{R} \rightarrow \mathbb{R}, f(x)=\sin(3x)$}
 %\item{$f: \mathbb{R} \rightarrow \mathbb{R}, f(x)=\cos(\frac{x}{4})$}
 %\item{$f: \mathbb{R} \rightarrow \mathbb{R}, f(x)=\sin(3x) + \cos(\frac{x}{4})$}
 %\item{$f: \mathbb{R} \rightarrow \mathbb{R}, f(x)=\lfloor x \rfloor$}
 \item $g(x)=x^3$, $f(x)=x-1$
\begin{solbox}
עבור $g(x)=x^3$, $f(x)=x-1$:
\[
,f(g(x))=x^3-1
\qquad
 ,g(f(x))=(x-1)^3
\]
\[
,f(f(x))=x-2
\qquad
 .g(g(x))=x^9
\]
\end{solbox}
 \item $g(x)=2^x$, $f(x)=x^2+3x$
\begin{solbox}
עבור $g(x)=2^x$, $f(x)=x^2+3x$:
\[
,f(g(x))=(2^x)^2+3\cdot2^x=4^x+3\cdot2^x
\]
\[
,g(f(x))=2^{x^2+3x}
\qquad
 ,f(f(x))=(x^2+3x)^2+3(x^2+3x)
\]
\[
.g(g(x))=2^{2^x}
\]
\end{solbox}

 \item $g(x)=2^2$, $f(x)=x^2$
\hint{שימו לב ש-$g$ היא פונקציה קבועה.}
\begin{solbox}
עבור $g(x)=2^2=4$, $f(x)=x^2$:
\[
.f(g(x))=16
\qquad
 ,g(f(x))=4
\qquad
 ,f(f(x))=x^4
\qquad
 ,g(g(x))=4
\]
\end{solbox}
 \item $g(x)=\ln(x)$, $f(x)=x^2$
\hint{שימו לב לתחום ההגדרה של כל הרכבה: $\ln$ מוגדר רק עבור ארגומנט חיובי.}
\begin{solbox}
עבור $g(x)=\ln x$, $f(x)=x^2$:
\[
,f(g(x))=(\ln x)^2\quad (x>0)
\]
\[
,g(f(x))=\ln(x^2)\quad (x\neq0)
\]
\[
.f(f(x))=x^4
\qquad
 ,g(g(x))=\ln(\ln x)\quad (x>1)
\]
\end{solbox}
% פונקציה זו נקראת `''פונקציית הערך השלם'', והיא מתאימה לכל מספר ממשי את המספר השלם המקסימלי הקטן או שווה לו.
%  \item{$f: \mathbb{R} \rightarrow \mathbb{R}, f(x)=\lfloor x \rfloor - x$}
\item $g(x)=2^x$, $f(x)=x^2$
\begin{solbox}
$f(g(x))=(2^x)^2=4^x, g(f(x))=2^{x^2},f(f(x))=x^4,g(g(x))=2^{(2^x)}$
\end{solbox}
\eMs
\end{enumerate}
