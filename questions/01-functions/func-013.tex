פונקציה
$\ f  $
נקראת חסומה אם קיים מספר ממשי
$M$
כך שלכל
$x$
בתחום ההגדרה
$\abs{f(x)}\leq M\ $.
   באופן שקול, $f$ חסומה אם  קיימים מספרים ממשיים $m<M$
   כך שלכל $x $,
   $  m\leq f(x) \leq M$ .\\
   \textbf{הוכיחו כי הפונקציות הבאות חסומות בתחום ההגדרה הטבעי שלהם:}

\begin{enumerate}\bMs{2}
\item
$f(x)=3+4\sin(2x)$
\hint{השתמשו בכך ש-$|\sin t|\leq1$ לכל $t$ ובאי-שוויון המשולש.}
\begin{solbox}
לכל $x$ מתקיים $|\sin(2x)|\leq1$, ולכן
\[
.|3+4\sin(2x)|\leq 3+4|\sin(2x)|\leq7.
\]
לכן הפונקציה חסומה.
\end{solbox}

\item $f(x)=\frac{x}{x^2+1}$
\hint{נסו לחסום את $|x|$ בעזרת $x^2+1$; אי-שוויון שנובע מביטוי מהצורה $(\ldots)^2\geq0$ יכול לעזור.}
\begin{solbox}
לכל $x$ מתקיים
\[
,\left|\frac{x}{x^2+1}\right|=\frac{|x|}{x^2+1}\leq \frac12
\]
כי
$x^2+1\geq2|x|$.
האי-שיוויון האחרון נובע מכך שמתקיים
$$.0\le (|a|-|b|)^2=a^2-2|ab|+b^2\Longrightarrow a^2+b^2\ge 2|ab|$$
לכן הפונקציה חסומה.
\end{solbox}
\item $f(x)=\sqrt{9-x^2}$
\begin{solbox}
תחום ההגדרה הוא $[-3,3]$, ובתחום זה
\[
,0\leq \sqrt{9-x^2}\leq3
\]
כי
$0\le 9-x^2\le 9$,
לכן הפונקציה חסומה.
\end{solbox}
\eMs
\end{enumerate}
